# TechStore — inicializacao local (execute na RAIZ do repositorio)
# Pre-requisitos: Docker + plugin compose (ou Node 20 + Mongo local p/ modo manual)

# 0. Configurar segredos (primeira vez; arquivos nao versionados)
cp -n .env.example .env
cp -n backend/.env.example backend/.env
# Edite .env (MONGO_PASSWORD) e backend/.env (MONGODB_URI com a MESMA senha,
# JWT_SECRET, ADMIN_PASSWORD) antes de subir.

# 1. Subir stack (mongo:7 + backend + frontend com healthchecks)
docker compose up -d --build
docker compose ps

# 2. Acompanhar saude (aguarde backend "healthy")
curl http://localhost:5010/health
docker logs ecommerce_backend --tail 5

# 3. Testes (unitarios + E2E contra a stack)
cd backend && npm test
E2E_BASE_URL=http://localhost:5010 \
E2E_ADMIN_EMAIL=admin@techstore.com.br \
E2E_ADMIN_PASSWORD="<ADMIN_PASSWORD do backend/.env>" \
npm run test:e2e

# 4. Derrubar (mantem dados do volume mongo_data; use -v p/ resetar)
docker compose down
