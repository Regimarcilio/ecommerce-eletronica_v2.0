# 1. Matar processos antigos
pkill node

# 2. Iniciar MongoDB (se não estiver rodando)
docker start ecommerce_mongodb 2>/dev/null || \
docker run -d \
  --name ecommerce_mongodb \
  -p 27017:27017 \
  -e MONGO_INITDB_ROOT_USERNAME=admin \
  -e MONGO_INITDB_ROOT_PASSWORD=***REMOVED*** \
  -e MONGO_INITDB_DATABASE=ecommerce \
  mongo:4.4

# 3. Instalar dependências do backend
cd backend
npm install

# 4. Iniciar backend
node server.js &

# 5. Aguardar backend
sleep 3

# 6. Iniciar frontend
cd ../frontend
python3 -m http.server 5500 &
